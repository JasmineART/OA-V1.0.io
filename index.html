Get API Access
JavaScript — REST API 
// Fetch AI readiness score for a project
const response = await fetch(
  'https://api.aoai.io/v1/projects/{id}/readiness',
  {
    method: 'GET',
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json'
    }
  }
);

const { score, kpis, alerts } =
  await response.json();

// Inject directly into your PM tool
updateDashboard({ score, kpis, alerts });
